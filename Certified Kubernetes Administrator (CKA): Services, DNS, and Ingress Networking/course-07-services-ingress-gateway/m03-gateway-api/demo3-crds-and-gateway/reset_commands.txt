# Cleanup: back to a cluster with no Gateway API and Traefik's Gateway provider off,
# so the CRD install, the helm upgrade and both Gateway applies are real changes.
# Provider off FIRST: a running provider whose CRDs vanish logs watch errors.
helm upgrade traefik traefik/traefik --version 41.4.0 --namespace traefik --reuse-values --set providers.kubernetesGateway.enabled=false
kubectl -n traefik rollout status deploy/traefik --timeout=180s
# Deleting the CRDs deletes every Gateway API object with them -- GatewayClass,
# Gateway, and any HTTPRoute a later demo left behind. Lab only.
kubectl delete -f https://github.com/kubernetes-sigs/gateway-api/releases/download/v1.6.1/standard-install.yaml --ignore-not-found --timeout=120s
# kubectl caches API discovery for hours. Without a refresh, Phase 1 prints
# "Unable to list gateway.networking.k8s.io/v1" instead of the line the narration
# describes. api-resources re-reads discovery and rewrites the cache.
kubectl api-resources > /dev/null
kubectl get gateways #allow-error
